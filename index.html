let draggedCard = null; // Add a new task to the "To Do" column function addTask() { const input = document.getElementById('taskInput'); const text = input.value.trim(); if (!text) return; const taskCard = createCardElement(text); document.getElementById('todo').appendChild(taskCard); input.value = ''; } // Create the task card HTML element with drag-and-drop capability function createCardElement(text) { const card = document.createElement('div'); card.className = 'task-card'; card.draggable = true; // Unique ID for dragging reference card.id = 'task-' + Date.now() + Math.random().toString(36).substr(2, 5); card.innerHTML = ` ${text} `; // Drag event listeners card.addEventListener('dragstart', (e) => { draggedCard = card; e.dataTransfer.setData('text/plain', card.id); }); card.addEventListener('dragend', () => { draggedCard = null; }); return card; } // Allow dropping items into columns function allowDrop(e) { e.preventDefault(); } // Handle the drop action into a specific column list function drop(e, columnId) { e.preventDefault(); const targetList = document.getElementById(columnId); if (draggedCard) { targetList.appendChild(draggedCard); } }
